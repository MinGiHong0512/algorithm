class Solution {
    public int solution(String[] board) {
        int answer = 1;
        
        int ocount = 0;
        int xcount = 0;
        for(int i = 0; i < 3; i++){
            for(int j = 0; j < 3; j++){
                char item = board[i].charAt(j);
                
                if(item == 'O'){
                    ocount += 1;
                }else if(item == 'X'){
                    xcount += 1;
                }
            }
        }
                
        if(xcount > ocount || ocount > xcount+1){
            return 0;
        }

        char[] tboard = new  char[9];
        
        for(int i = 0; i < 9; i++){
            tboard[i] = board[i/3].charAt(i%3);
        }

        
        int[][] Bingo = {
            {0,1,2},
            {3,4,5},
            {6,7,8},
            {0,3,6},
            {1,4,7},
            {2,5,8},
            {0,4,8},
            {2,4,6},
        };
        
        int oBingo = 0;
        int xBingo = 0;
        
        for(int i = 0; i < Bingo.length; i++){
            int a = Bingo[i][0];
            int b = Bingo[i][1];
            int c = Bingo[i][2];
            
            if(tboard[a] == tboard[b] && tboard[b] == tboard[c] && tboard[a] != '.'){
                if(tboard[a] == 'O'){
                    oBingo += 1;
                }else if(tboard[a] == 'X'){
                    xBingo += 1;
                }
            }
        }
        
        
        if(oBingo > 0 && xBingo > 0){
            answer = 0;
        }else if(oBingo > 0 && ocount - 1 != xcount){
            answer = 0;
        }else if(xBingo > 0 && ocount != xcount){
            answer = 0;
        }
        
        return answer;
    }
}
