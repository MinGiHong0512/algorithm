class Solution {
    public int solution(int[] arr) {
        int answer = 1;
        
        for(int i = 0; i < arr.length; i++){
            answer = LCM(answer, arr[i]);
        }
        
        
        return answer;
        
        //GCD LCM
        
    }
    
    public int GCD(int a, int b){
        while(true){
            if(a <= b){
                int temp = 0;
                temp = a;
                a = b;
                b = temp;
            }
            
            if(b == 0){
                return a;
            }
            
            a %= b;
        }
    }
    public int LCM(int a, int b){
        return (a * b) / GCD(a,b);
    }
}
