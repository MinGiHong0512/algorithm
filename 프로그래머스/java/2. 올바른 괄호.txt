import java.util.Stack;
class Solution {
    boolean solution(String s) {
      	Stack<Character> st = new Stack<>();
        boolean answer = true;
        for (var item : s.toCharArray()) {
			if(item == '(') st.push('(');
			else {
				if(st.isEmpty()) return false;
				st.pop();
			}
		}
        if(st.size() > 0 ) answer = false;
        return answer;
    }
}
