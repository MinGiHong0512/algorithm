import java.util.*;

class Solution {
    public int solution(int[] players, int m, int k) {
        int answer = 0;
        
        //0 1 2 3 4 5 6 .... 21 22 23
        //각 숫자는 시작시간(ex: 0이면 0~1를 나타냄)
        
        ArrayList<Integer> timeList = new ArrayList<>();
        
                
        for(int i = 0; i < 24; i++){
            int n = 0;
            
            for(int j = 0; j < timeList.size(); j++){
                if(i - timeList.get(j) < k){
                    n += 1;
                }
            }
            
            // System.out.println("index"+i+"  "+n*m+"    " + (players[i] - m) +"      " + players[i] / m);
            
            if(n*m <=players[i] - m){
                for(int j = 0; j < players[i] / m -n; j++){
                    timeList.add(i);
                }         
            }
            
        }
        
        // for(int i = 0; i < timeList.size(); i++){
        //     System.out.println(timeList.get(i));
        // }
        
        answer = timeList.size();
        
        return answer;
    }
}
