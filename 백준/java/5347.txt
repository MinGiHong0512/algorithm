import java.util.Scanner;

public class Main {
	public static void main(String[] args) {
		Scanner scan = new Scanner(System.in);

		int n = scan.nextInt();

		for (int i = 0; i < n; i++) {
			long a = scan.nextLong();
			long b = scan.nextLong();

			long mul = a * b;

			while (b != 0) {
				long temp = a % b;
				a = b;
				b = temp;
			}

			System.out.println(mul / a);
		}
	}
}
