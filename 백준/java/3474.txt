import java.util.Scanner;

public class Main {
	public static void main(String[] args) {
		Scanner scan = new Scanner(System.in);

		int n = scan.nextInt();
		
		for(int i = 0; i < n; i++) {
			int fn = scan.nextInt();
			
			int answer = 0;
			
			while(fn >= 5) {
				fn /= 5;
				answer += fn;
			}
			
			System.out.println(answer);
			
		}
	}
}
