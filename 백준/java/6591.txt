import java.util.Scanner;

public class Main {
	public static void main(String[] args) {
		Scanner scan = new Scanner(System.in);

        while (true) {
            int a = scan.nextInt();
            int b = scan.nextInt();

            if (a == 0 && b == 0) break;

            b = Math.min(b, a - b);

            long result = 1;

            for (int i = 1; i <= b; i++) {
                result = result * (a - b + i) / i;
            }

            System.out.println(result);
        }
	}
}
